# CSG Agents: local full-system run (PowerShell)
# Run these commands from the repository root.

# One-time setup
python -m venv .venv
.venv\Scripts\Activate.ps1
pip install -e Research_Agent -e Outreach_Agent -e Server
Set-Location Dashboard
npm install
Set-Location ..

# Terminal 1: backend (Research + Outreach)
python -m uvicorn app:app --app-dir Server --reload --reload-dir Research_Agent/src --reload-dir Outreach_Agent/src --reload-dir Server --port 8000

# Terminal 2: Dashboard
Set-Location Dashboard
npm run dev

# Open:
# Dashboard:       http://localhost:5173
# Research API:    http://127.0.0.1:8000/research/docs
# Outreach API:    http://127.0.0.1:8000/outreach/docs
# Combined API:    http://127.0.0.1:8000/docs
